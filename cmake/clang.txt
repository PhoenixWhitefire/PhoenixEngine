# settings formatted for G++

set(PHX_TRACY_LIB_PATH Vendor/tracy/${PHX_BUILD_TYPE}/TracyClient)

target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:ReleaseTSan>:-O3>)
target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:ReleaseAUSan>:-O3>)
target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:Release>:-O3 -g>)
target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:DebugTSan>:-O0>)
target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:DebugTSan>:-g3>)
target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:Debug>:-O0>)
target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:Debug>:-g3>)
target_compile_options(PhoenixEngine PRIVATE -MP)

target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:DebugTSan>:-Wno-error=tsan>) # atomic_thread_fence in tracy_concurrentqueue.h

target_compile_options(PhoenixEngine PRIVATE
    -Wall -Wpedantic
    -Werror

    -Wno-nested-anon-types
    -Wno-gnu-zero-variadic-macro-arguments
)

# including Vendor/luau/Ast/include/luau/Ast.h
set_property(SOURCE src/impl/script/ScriptEngine.cpp PROPERTY COMPILE_OPTIONS "-Wno-unused-parameter")

target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:ReleaseTSan>:-fsanitize=thread>)
target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:DebugTSan>:-fsanitize=thread>)
target_compile_options(PhoenixEngine PRIVATE $<$<CONFIG:Debug>:-fsanitize=address,undefined>)
target_compile_options(PhoenixEngine PRIVATE $<$<OR:$<CONFIG:Debug>,$<CONFIG:ReleaseAUSan>>:-fsanitize=address,undefined>)

target_link_options(PhoenixEngine PRIVATE $<$<CONFIG:ReleaseTSan>:-flto -fsanitize=thread>)
target_link_options(PhoenixEngine PRIVATE $<$<CONFIG:DebugTSan>:-fsanitize=thread>)
target_link_options(PhoenixEngine PRIVATE $<$<CONFIG:Release>:-flto>)
target_link_options(PhoenixEngine PRIVATE $<$<CONFIG:Debug>:-fsanitize=address,undefined>)
target_link_options(PhoenixEngine PRIVATE $<$<OR:$<CONFIG:Debug>,$<CONFIG:ReleaseAUSan>>:-fsanitize=address,undefined>)

target_link_libraries(PhoenixEngine PRIVATE stdc++)

add_compile_options($<$<CONFIG:ReleaseTSan>:-O3>)
add_compile_options($<$<CONFIG:ReleaseAUSan>:-O3>)
add_compile_options($<$<CONFIG:DebugTSAN>:-fsanitize=thread>)
add_compile_options($<$<CONFIG:DebugTSan>:-O0>)
add_compile_options($<$<CONFIG:DebugTSan>:-g3>)
add_compile_options($<$<CONFIG:Release>:-O3>)
add_compile_options($<$<CONFIG:Debug>:-fsanitize=address,undefined>)
add_compile_options($<$<CONFIG:Debug>:-O0>)
add_compile_options($<$<CONFIG:Debug>:-g3>)

add_compile_options(-MP)

add_link_options($<$<CONFIG:ReleaseTSan>:-fsanitize=thread>)
add_link_options($<$<CONFIG:ReleaseTSan>:-flto>)
add_link_options($<$<CONFIG:DebugTSan>:-fsanitize=thread>)
add_link_options($<$<CONFIG:Release>:-flto>)
add_link_options($<$<CONFIG:Debug>:-fsanitize=address,undefined>)

set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -fuse-ld=lld")
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -fuse-ld=lld")

if (NOT CMAKE_OBJCOPY)
    set(CMAKE_OBJCOPY objcopy)
endif()

set(TARGET_FILE_NAME "$<TARGET_FILE:PhoenixEngine>")

add_custom_command(TARGET PhoenixEngine POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} --only-keep-debug;${TARGET_FILE_NAME} ${TARGET_FILE_NAME}.debug
    COMMAND ${CMAKE_OBJCOPY} --strip-debug ${TARGET_FILE_NAME}

    COMMAND ${CMAKE_OBJCOPY} --add-gnu-debuglink=${TARGET_FILE_NAME}.debug ${TARGET_FILE_NAME}

    VERBATIM
)
